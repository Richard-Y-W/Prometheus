cmake_minimum_required(VERSION 3.24)
project(Prometheus VERSION 0.1.0 LANGUAGES C CXX)
include(CTest)
option(PROMETHEUS_BUILD_DESKTOP "Build the Qt desktop shell" ON)
option(PROMETHEUS_BUILD_INTEGRITY "Build the independent integrity verifier" OFF)
option(PROMETHEUS_BUILD_EXECUTION "Build the Qt-free execution library" ON)
option(PROMETHEUS_BUILD_RUN_STORE "Build the Qt-free execution object store" ON)
option(PROMETHEUS_BUILD_REPLAY "Build exact offline replay support and CLI" ON)
option(PROMETHEUS_ENABLE_OCCT "Enable Open Cascade CAD adapter" OFF)
option(PROMETHEUS_ENABLE_MUJOCO "Enable MuJoCo adapter" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_subdirectory(desktop/core)
if(PROMETHEUS_BUILD_INTEGRITY OR PROMETHEUS_BUILD_EXECUTION OR PROMETHEUS_BUILD_RUN_STORE OR PROMETHEUS_BUILD_REPLAY OR PROMETHEUS_BUILD_DESKTOP)
  add_subdirectory(desktop/integrity)
endif()
if(PROMETHEUS_BUILD_EXECUTION)
  add_subdirectory(desktop/execution)
endif()
if(PROMETHEUS_BUILD_RUN_STORE)
  add_subdirectory(desktop/run_store)
endif()
if(PROMETHEUS_BUILD_REPLAY)
  if(NOT PROMETHEUS_BUILD_EXECUTION OR NOT PROMETHEUS_BUILD_RUN_STORE)
    message(FATAL_ERROR "Replay requires execution and run-store targets")
  endif()
  add_subdirectory(desktop/replay)
endif()
if(PROMETHEUS_ENABLE_OCCT)
  find_package(OpenCASCADE REQUIRED COMPONENTS FoundationClasses ModelingData ModelingAlgorithms Visualization ApplicationFramework DataExchange)
  add_subdirectory(desktop/cad)
endif()
if(PROMETHEUS_BUILD_DESKTOP)
  if(NOT PROMETHEUS_BUILD_RUN_STORE OR NOT PROMETHEUS_BUILD_EXECUTION OR
     NOT PROMETHEUS_BUILD_REPLAY)
    message(FATAL_ERROR
      "Desktop execution requires execution, run-store, and replay support"
    )
  endif()
  find_package(Qt6 6.5 REQUIRED COMPONENTS Quick QuickControls2 Quick3D Concurrent Network)
  if(Qt6_FOUND)
    add_subdirectory(desktop/app)
  endif()
endif()
