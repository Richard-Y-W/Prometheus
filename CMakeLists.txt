cmake_minimum_required(VERSION 3.24)
project(Prometheus VERSION 0.1.0 LANGUAGES C CXX)
include(CTest)
option(PROMETHEUS_BUILD_DESKTOP "Build the Qt desktop shell" ON)
option(PROMETHEUS_BUILD_INTEGRITY "Build the independent integrity verifier" OFF)
option(PROMETHEUS_ENABLE_OCCT "Enable Open Cascade CAD adapter" OFF)
option(PROMETHEUS_ENABLE_MUJOCO "Enable MuJoCo adapter" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_subdirectory(desktop/core)
if(PROMETHEUS_BUILD_INTEGRITY)
  find_package(Qt6 6.5 REQUIRED COMPONENTS Core)
  add_subdirectory(desktop/integrity)
endif()
if(PROMETHEUS_ENABLE_OCCT)
  find_package(OpenCASCADE REQUIRED COMPONENTS FoundationClasses ModelingData ModelingAlgorithms Visualization ApplicationFramework DataExchange)
  add_subdirectory(desktop/cad)
endif()
if(PROMETHEUS_BUILD_DESKTOP)
  find_package(Qt6 6.5 COMPONENTS Quick QuickControls2 Quick3D Concurrent Network QUIET)
  if(Qt6_FOUND)
    add_subdirectory(desktop/app)
  else()
    message(WARNING "Qt 6.5+ not found: native desktop shell disabled. Configure with Qt6_DIR or use PROMETHEUS_BUILD_DESKTOP=OFF.")
  endif()
endif()
